{{- /* Base template. The original site walks pages in a single cycle: Introduction -> 1-00 -> ... -> 5-00 -> Introduction. Previous/next are resolved explicitly from the weight order so that the wrap-around back to the home page is reproduced exactly. */ -}} {{- $isHome := eq .RelPermalink "/" -}} {{- $rp := .Site.RegularPages -}} {{- $prev := false -}} {{- $next := false -}} {{- if $isHome -}} {{- $next = index $rp 0 -}} {{- else -}} {{- $idx := -1 -}} {{- range $i, $p := $rp -}} {{- if eq $p.RelPermalink $.RelPermalink -}}{{- $idx = $i -}}{{- end -}} {{- end -}} {{- $prev = $.Site.Home -}} {{- $next = $.Site.Home -}} {{- if gt $idx 0 -}}{{- $prev = index $rp (sub $idx 1) -}}{{- end -}} {{- if lt $idx (sub (len $rp) 1) -}}{{- $next = index $rp (add $idx 1) -}}{{- end -}} {{- end -}} {{- $ctx := dict "page" . "prev" $prev "next" $next "isHome" $isHome -}} {{ partial "head.html" $ctx }} {{ partial "header.html" $ctx }} {{- block "main" $ctx }}{{ end -}} {{ partial "footer.html" $ctx }}
